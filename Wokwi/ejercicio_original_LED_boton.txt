// Sketch autónomo para probar botones, LED y buzzer en Wokwi. No prueba BLE, MQTT o Firebase.
#define LED_PIN 4
#define BUZZER_PIN 5
#define BUTTON_PIN 6
void setup(){ pinMode(LED_PIN,OUTPUT); pinMode(BUZZER_PIN,OUTPUT); pinMode(BUTTON_PIN,INPUT_PULLUP); }
void loop(){
 bool press=digitalRead(BUTTON_PIN)==LOW;
 digitalWrite(LED_PIN,press?HIGH:LOW);
 digitalWrite(BUZZER_PIN,press?HIGH:LOW);
 delay(20);
}
